# Call-control signaling (ring + receive) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ring the callee's phone from Linux by invoking `sendRequestCall` (`voicecall/request`) via CDP; on answer, receive+decrypt the peer's real media through the 2.2 `MediaSession`; `endCall` on exit.

**Architecture:** Generalize the CDP invoker to call any VoiceCall API method by name; add pure call-control helpers that build the `sendRequestCall`/`sendEndCall` args from a config; add an operator `live-call.js` that runs requestCall → handshake → ring → MediaSession → endCall.

**Tech Stack:** Node.js + CDP (reuse `tools/zcall-signaling`), 2.1 handshake + 2.2 MediaSession.

**Spec:** `docs/superpowers/specs/2026-07-13-zcall-sp2-2.3-call-control-signaling-design.md`

## Global Constraints

- **Boundary:** operator's own account / machine / **own phone** only — this step RINGS a phone. `sessId`/relay-addresses not committed; live output redacted; `endCall` always runs on exit.
- **No `Co-Authored-By` / AI-attribution** anywhere. Commit only when asked.
- Signaling arg order (from the bundle): `sendRequestCall(calleeId, rtcpAddress, rtpAddress, codec, extendData, session, callId)`; `sendEndCall(uidTo, callId)`.

'use strict';
// Live signaling via CDP: reach the app's own webpack module registry and invoke any VoiceCall
// API method (requestCall, sendRequestCall, sendEndCall, …) — reusing the app's auth, zpw signing
// and zpw decode. Needs Zalo launched with --remote-debugging-port. Own account / own phone only.

// Pure: the JS expression evaluated in the page. Self-locates the module/object exposing
// `[method]` (robust to minified module-id changes) via the webpack-4 `webpackJsonp` require grab,
// then calls api[method](...args).
function buildVoiceCallExpr({ method, args }) {
  const m = JSON.stringify(String(method));
  const a = JSON.stringify(args || []);
  return '(async()=>{try{' +
    'var J=window.webpackJsonp; if(!J||typeof J.push!=="function") return JSON.stringify({err:"no-webpackJsonp"});' +
    'var BID=987654321; if(!window.__wreq){J.push([[BID],{[BID]:function(m,e,r){window.__wreq=r;}},[[BID]]]);}' +
    'var req=window.__wreq; if(!req||!req.c) return JSON.stringify({err:"no-require-cache"});' +
    'var M=' + m + '; var api=null; var ids=Object.keys(req.c);' +
    'for(var i=0;i<ids.length;i++){var ex;try{ex=req.c[ids[i]]&&req.c[ids[i]].exports;}catch(_){continue;} if(!ex)continue;' +
    'if(typeof ex[M]==="function"){api=ex;break;} if(ex.default&&typeof ex.default[M]==="function"){api=ex.default;break;}}' +
    'if(!api) return JSON.stringify({err:"no-"+M+"-module"});' +
    'var args=' + a + ';' +
    'var resp=await api[M].apply(api, args);' +
    'return JSON.stringify(resp===undefined?{ok:true}:resp);' +
    '}catch(e){return JSON.stringify({err:String(e&&e.message||e)});}})()';
}

// Back-compat wrapper for the step-1 requestCall path.
function buildInvokeExpr({ calleeId, callId, type }) {
  return buildVoiceCallExpr({ method: 'requestCall', args: [String(calleeId), callId, '[]', Number(type)] });
}

// Live: evaluate a method invocation in the index.html page target, return the parsed result.
async function invokeVoiceCall({ port = 9222, method, args } = {}) {
  if (!method) throw new Error('cdp-invoke: method required');
  const targets = await (await fetch('http://127.0.0.1:' + port + '/json')).json();
  const page = targets.find((t) => t.type === 'page' && (t.url || '').includes('index.html')) || targets.find((t) => t.type === 'page');
  if (!page) throw new Error('cdp-invoke: no page target — launch Zalo with --remote-debugging-port=' + port + ' (ZALO_REMOTE_DEBUG=1)');
  const ws = new WebSocket(page.webSocketDebuggerUrl);
  await new Promise((res, rej) => { ws.onopen = res; ws.onerror = () => rej(new Error('cdp-invoke: WS connect failed')); });
  let id = 0; const pend = new Map();
  ws.onmessage = (ev) => { const m = JSON.parse(ev.data); if (m.id && pend.has(m.id)) { pend.get(m.id)(m); pend.delete(m.id); } };
  const send = (cmd, params) => new Promise((res) => { const i = ++id; pend.set(i, res); ws.send(JSON.stringify({ id: i, method: cmd, params: params || {} })); });
  await send('Runtime.enable');
  const r = await send('Runtime.evaluate', { expression: buildVoiceCallExpr({ method, args }), returnByValue: true, awaitPromise: true });
  ws.close();
  if (r.result && r.result.exceptionDetails) throw new Error('cdp-invoke: eval exception ' + JSON.stringify(r.result.exceptionDetails).slice(0, 200));
  const val = r.result && r.result.result && r.result.result.value;
  const obj = JSON.parse(val);
  if (obj.err) throw new Error('cdp-invoke: ' + obj.err + (obj.keys ? ' keys=' + JSON.stringify(obj.keys) : ''));
  return obj;
}

async function invokeRequestCall({ port = 9222, calleeId, callId, type = 1 } = {}) {
  if (!calleeId) throw new Error('cdp-invoke: calleeId required');
  return invokeVoiceCall({ port, method: 'requestCall', args: [String(calleeId), callId, '[]', Number(type)] });
}

module.exports = { buildVoiceCallExpr, buildInvokeExpr, invokeVoiceCall, invokeRequestCall };
```

'use strict';
// Call-control signaling helpers (SP2 2.3): ring the callee (voicecall/request) and end the call
// (voicecall/endcall) by invoking the app's own API methods via CDP. Own phone only.
const { invokeVoiceCall } = require('./cdp-invoke.js');

// Resolve the (rtp, rtcp) media addresses to advertise in sendRequestCall.
function pickAddr(config, relayAddr, source) {
  if (source === 'server0') {
    const s = (config.servers && config.servers[0]) || {};
    return { rtp: s.rtpaddr || '', rtcp: s.rtcpaddr || '' };
  }
  if (source === 'relay') {
    const a = relayAddr ? (relayAddr.ip + '|' + relayAddr.port) : '';
    return { rtp: a, rtcp: a };
  }
  return { rtp: config.rtpIP || '', rtcp: config.rtcpIP || '' }; // default 'config'
}

// Positional args for sendRequestCall(calleeId, rtcpAddress, rtpAddress, codec, extendData, session, callId).
function buildRingArgs({ config, callId, relayAddr, addrSource = 'config' }) {
  const { rtp, rtcp } = pickAddr(config, relayAddr, addrSource);
  return [String(config.toId), rtcp, rtp, '[]', '', config.sessId, callId];
}

function ring({ callId, config, relayAddr, addrSource = 'config', port = 9222 }) {
  return invokeVoiceCall({ port, method: 'sendRequestCall', args: buildRingArgs({ config, callId, relayAddr, addrSource }) });
}

function endCall({ uidTo, callId, port = 9222 }) {
  return invokeVoiceCall({ port, method: 'sendEndCall', args: [String(uidTo), callId] });
}

module.exports = { pickAddr, buildRingArgs, ring, endCall };
```

```js
'use strict';
// Operator-run live CONNECTED call on Linux (own account / own phone). requestCall -> InitZRTP
// handshake -> sendRequestCall (RINGS your phone) -> MediaSession receives the peer's real media
// on answer -> endCall on exit. Prints a REDACTED summary. Payload sent is synthetic (opus = step 3).
const { invokeRequestCall } = require('../zcall-signaling/cdp-invoke.js');
const { parseConfig, srtpMasterKey } = require('../zcall-signaling/requestcall.js');
const { ring, endCall } = require('../zcall-signaling/call-control.js');
const { handshake } = require('./handshake.js');
const { MediaSession } = require('./media-session.js');

function mask(s) { return String(s).replace(/[0-9A-Fa-f]+/g, '***'); }
function flag(name, def) { const i = process.argv.indexOf(name); return i >= 0 && process.argv[i + 1] ? process.argv[i + 1] : def; }

async function main() {
  const calleeId = process.argv[2];
  if (!calleeId) throw new Error('usage: node tools/zcall-media/live-call.js <calleeId> [--addr config|server0|relay] [--dur 20000]');
  const addrSource = flag('--addr', 'config');
  const durationMs = Number(flag('--dur', '20000'));

  const callId = Math.floor(Math.random() * 1e9);
  const config = parseConfig(JSON.stringify(await invokeRequestCall({ calleeId, callId, type: 1 })));
  const key = srtpMasterKey(config.sessId);

  const replies = await handshake({ fromId: config.fromId, toId: config.toId, callId, sessId: config.sessId, servers: config.servers });
  const relayAddr = replies.length ? replies[0].relayAddr : null;

  const s = new MediaSession({ key, ssrc: config.fromId, relayAddr: relayAddr || { host: '127.0.0.1', port: 1 } });
  let inOk = 0, inFail = 0, seenFlow = null;
  s.on('media', (m) => { inOk++; if (!seenFlow) seenFlow = m.flowToken; });
  s.on('authfail', () => { inFail++; });

  let rang = '?';
  try {
    await ring({ callId, config, relayAddr, addrSource });
    rang = 'ok';
  } catch (e) { rang = 'err:' + e.message; }

  try {
    let timer = null;
    if (relayAddr) {
      await new Promise((res) => s.bind(res));
      const filler = Buffer.alloc(80, 0);
      timer = setInterval(() => { try { s.send(filler); } catch (_) {} }, 20);
    }
    await new Promise((res) => setTimeout(res, durationMs));
    if (timer) clearInterval(timer);
  } finally {
    s.close();
    try { await endCall({ uidTo: config.toId, callId }); } catch (_) {}
  }

  console.log('[live-call] rang ' + rang + '  addrSource ' + addrSource);
  console.log('[live-call] inboundAuthOk ' + inOk + '  authfail ' + inFail);
  console.log('[live-call] inbound flowToken ' + (seenFlow ? mask(seenFlow.toString('hex')) : '<none>'));
  console.log('[live-call] relay ' + (relayAddr ? mask((relayAddr.ip || '') + '|' + (relayAddr.port || '')) : '<none>'));
  console.error('[live-call] ' + (inOk ? 'OK — connected call, decrypted peer real media on Linux' : 'no inbound (ring ' + rang + '; try --addr server0|relay, and answer the phone)'));
  process.exit(inOk ? 0 : 1);
}

if (require.main === module) main().catch((e) => { console.error('[live-call] FAILED:', e.message); process.exit(1); });
module.exports = { main };
```
